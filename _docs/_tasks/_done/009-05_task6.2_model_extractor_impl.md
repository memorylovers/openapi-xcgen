### ModelExtractor関数

```typescript
// src/transformer/extractors/model-extractor.ts
import type { OpenAPIDocument } from "../../types";
import type { IRModel, IRProperty, IRType } from "../../types/ir";

/**
 * Extract models from OpenAPI components.schemas
 * @param doc - OpenAPI document to extract from
 * @returns Array of IRModel definitions
 */
export function extractModels(doc: OpenAPIDocument): IRModel[] {
  if (!doc.components?.schemas) {
    return [];
  }

  const models: IRModel[] = [];
  
  for (const [name, schema] of Object.entries(doc.components.schemas)) {
    if (typeof schema === "object" && "type" in schema && schema.type === "object") {
      models.push(extractModel(name, schema));
    }
  }
  
  return models;
}

function extractModel(name: string, schema: any): IRModel {
  return {
    name,
    description: schema.description,
    properties: extractProperties(schema),
    imports: [],
  };
}

function extractProperties(schema: any): IRProperty[] {
  if (!schema.properties) {
    return [];
  }

  const required = new Set(schema.required || []);
  const properties: IRProperty[] = [];

  for (const [name, prop] of Object.entries(schema.properties)) {
    properties.push({
      name,
      type: resolveType(prop),
      required: required.has(name),
      description: (prop as any).description,
    });
  }

  return properties;
}

function resolveType(schema: any): IRType {
  // 型解決ロジック
  if ("$ref" in schema) {
    return {
      kind: "ref",
      name: extractRefName(schema.$ref),
    };
  }

  if (schema.type === "array") {
    return {
      kind: "array",
      items: resolveType(schema.items),
    };
  }

  if (schema.type === "object") {
    return {
      kind: "map",
      key: { kind: "primitive", type: "string" },
      value: { kind: "any" },
    };
  }

  // プリミティブ型
  return {
    kind: "primitive",
    type: mapPrimitiveType(schema.type, schema.format),
  };
}

function extractRefName(ref: string): string {
  const parts = ref.split("/");
  return parts[parts.length - 1];
}

function mapPrimitiveType(type: string, format?: string): string {
  // 型マッピングロジック
  if (type === "integer") return "int";
  if (type === "number") return "double";
  if (type === "string") {
    if (format === "date-time") return "DateTime";
    if (format === "date") return "Date";
    if (format === "uuid") return "UUID";
    return "string";
  }
  if (type === "boolean") return "boolean";
  return "any";
}
```

### Transform関数の更新

```typescript
// src/transformer/transformer.ts
import { extractMetadata } from "./extractors";
import { extractModels } from "./extractors";

export function transform(doc: OpenAPIDocument): XcgenIR {
  return {
    metadata: extractMetadata(doc),
    models: extractModels(doc),
    enums: [],
    unions: [],
    services: [],
    servers: [],
  };
}
```

### Test 1: 基本的なモデル抽出

```typescript
it("should extract a simple model", () => {
  const doc = {
    openapi: "3.0.0",
    info: { title: "Test", version: "1.0.0" },
    components: {
      schemas: {
        User: {
          type: "object",
          properties: {
            id: { type: "integer" },
            name: { type: "string" },
          },
          required: ["id"],
        },
      },
    },
  };

  const ir = transform(doc);
  
  expect(ir.models).toHaveLength(1);
  expect(ir.models[0].name).toBe("User");
  expect(ir.models[0].properties).toHaveLength(2);
});
```

### Test 2: Required/Optionalプロパティ

```typescript
it("should correctly identify required and optional properties", () => {
  const doc = {
    openapi: "3.0.0",
    info: { title: "Test", version: "1.0.0" },
    components: {
      schemas: {
        User: {
          type: "object",
          properties: {
            id: { type: "integer" },
            name: { type: "string" },
            email: { type: "string" },
          },
          required: ["id", "name"],
        },
      },
    },
  };

  const ir = transform(doc);
  const user = ir.models[0];
  
  expect(user.properties[0].required).toBe(true);  // id
  expect(user.properties[1].required).toBe(true);  // name
  expect(user.properties[2].required).toBe(false); // email
});
```

### Test 3: 型解決

```typescript
it("should resolve different types correctly", () => {
  const doc = {
    openapi: "3.0.0",
    info: { title: "Test", version: "1.0.0" },
    components: {
      schemas: {
        Product: {
          type: "object",
          properties: {
            id: { type: "integer" },
            price: { type: "number" },
            name: { type: "string" },
            active: { type: "boolean" },
            tags: {
              type: "array",
              items: { type: "string" },
            },
            category: {
              $ref: "#/components/schemas/Category",
            },
          },
        },
      },
    },
  };

  const ir = transform(doc);
  const product = ir.models[0];
  
  expect(product.properties[0].type).toEqual({ kind: "primitive", type: "int" });
  expect(product.properties[1].type).toEqual({ kind: "primitive", type: "double" });
  expect(product.properties[2].type).toEqual({ kind: "primitive", type: "string" });
  expect(product.properties[3].type).toEqual({ kind: "primitive", type: "boolean" });
  expect(product.properties[4].type).toEqual({
    kind: "array",
    items: { kind: "primitive", type: "string" },
  });
  expect(product.properties[5].type).toEqual({ kind: "ref", name: "Category" });
});
```
